prontera,155,184,4<TAB>script<TAB>Absinthe Fairy#aft<TAB>4_F_KAFRA1,{
	.@c$ = "[^2E8B57Absinthe Fairy^000000]";
	mes .@c$;
	mes "Mmm... you smell of iron and ambition.";
	mes "Care for a glass of the Green Hour?";
	switch (select("Enter the Absinthe Tower:Tower status & records:Absinthe Token shop:How does this place work?:Leave")) {
	case 1:
		mes .@c$;
		if ($@abt_run) {
			mes "The Green Hour is already underway.";
			mes "Party of ^FF0000"+$@abt_leader$+"^000000 is on floor "+$@abt_floor+".";
			close;
		}
		.@cost  = getvariableofnpc(.cost, "AbsintheCtl#aft");
		.@minlv = getvariableofnpc(.min_lv, "AbsintheCtl#aft");
		.@map$  = getvariableofnpc(.map$, "AbsintheCtl#aft");
		if (BaseLevel < .@minlv) {
			mes "Come back at base level "+.@minlv+" or higher, little sprout.";
			close;
		}
		.@pid = getcharid(1);
		if (.@pid && getpartyleader(.@pid, 2) != getcharid(3)) {
			mes "Only your party leader may order for the table.";
			close;
		}
		mes "Choose your glass. Harder glasses make monsters stronger and rewards bigger.";
		next;
		.@d = select("Sugar Cube (x1 HP / x1 rewards):Louche (x1.5 HP / x1.5 rewards):Wormwood (x2.5 HP, x1.5 ATK / x2.5 rewards):The Green Hour (x4 HP, x2 ATK / x4 rewards)") - 1;
		setarray .@hp,  100, 150, 250, 400;
		setarray .@atk, 100, 125, 150, 200;
		setarray .@rw,  100, 150, 250, 400;
		mes .@c$;
		mes "Entry fee: ^FF0000"+.@cost+"z^000000. Your party is warped in. Time limit: "+(getvariableofnpc(.limit, "AbsintheCtl#aft") / 60)+" minutes.";
		mes "Party members must stand on this map to be taken.";
		if (select("Pour the glass:Never mind") == 2) close;
		if ($@abt_run) close; // someone was faster
		if (Zeny < .@cost) {
			mes "Your purse is lighter than my wings.";
			close;
		}
		Zeny -= .@cost;
		$@abt_run = 1;
		$@abt_hp  = .@hp[.@d];
		$@abt_atk = .@atk[.@d];
		$@abt_rw  = .@rw[.@d];
		$@abt_leader$ = strcharinfo(0);
		close2;
		if (.@pid) warpparty .@map$, 0, 0, .@pid, strnpcinfo(4);
		else warp .@map$, 0, 0;
		donpcevent "AbsintheCtl#aft::OnStart";
		end;
	case 2:
		mes .@c$;
		if ($@abt_run) mes "Run in progress: floor ^FF0000"+$@abt_floor+"^000000, leader "+$@abt_leader$+".";
		else mes "The tower is ^00AA00open^000000.";
		mes "Your best floor: ^0000FF"+abt_best+"^000000";
		mes "Server record: floor ^0000FF"+$abt_rec+"^000000 by "+$abt_rec_name$;
		mes "Full clears on this server: "+$abt_clears;
		mes "Absinthe Tokens in your bag: "+countitem(32100);
		close;
	case 3:
		.@n = 0;
		.@m$ = "";
		for (.@i = 0; .@i < getarraysize(.shop_id); .@i++) {
			if (getitemname(.shop_id[.@i]) == "null") continue;
			.@idx[.@n] = .@i;
			.@n++;
			.@m$ += getitemname(.shop_id[.@i])+" x"+.shop_amt[.@i]+" ("+.shop_cost[.@i]+" tokens):";
		}
		.@m$ += "Back";
		mes .@c$;
		mes "You hold ^0000FF"+countitem(32100)+"^000000 tokens.";
		.@s = select(.@m$) - 1;
		if (.@s >= .@n) close;
		.@k = .@idx[.@s];
		if (countitem(32100) < .shop_cost[.@k]) {
			mes "Not enough tokens, darling.";
			close;
		}
		if (checkweight(.shop_id[.@k], .shop_amt[.@k]) == 0) {
			mes "You carry too much already.";
			close;
		}
		delitem 32100, .shop_cost[.@k];
		getitem .shop_id[.@k], .shop_amt[.@k];
		mes "Enjoy.";
		close;
	case 4:
		mes .@c$;
		mes "100 floors. Every floor you must clear every monster.";
		mes "Floors ending in 5 have a bonus MVP, floors ending in 10 have a main MVP, floor 100 has three.";
		mes "Every third floor has a Gilded mini boss.";
		next;
		mes .@c$;
		mes "Monsters here carry alchemical drops: Philosopher's Stone Fragments, the four elemental stones, Old Card Albums, and my Mini Boss / MVP Card Albums.";
		mes "Rates follow the server card rate and grow with floor depth and glass strength.";
		mes "Each floor clear pays Absinthe Tokens, EXP and milestone boxes. Spend tokens here, transmute fragments with the Alchemist.";
		close;
	case 5:
		close;
	}
	end;

OnInit:
	setarray .shop_id,   7619, 7620, 603, 617, 616, 12103, 607, 32101, 32102, 32103, 32104, 32105;
	setarray .shop_amt,     5,    5,   1,   1,   1,     1,   5,     1,     1,     1,     1,     3;
	setarray .shop_cost,   10,   10,  15,  25,  60,    30,  20,     8,    40,    45,    90,    30;
	end;
}

-<TAB>script<TAB>AbsintheCtl#aft<TAB>-1,{
OnInit:
	// ---------------- CONFIG ----------------
	.map$ = "quiz_01";     // existing map; keep mapflags at bottom in sync
	.limit = 3600;         // seconds per run
	.min_lv = 60;
	.cost = 20000;         // zeny entry fee
	.maxfloor = 100;
	.max_wave = 40;
	// drop base chances are in 1/10000 and are multiplied by:
	//  (100 + floor*2)% * glass reward% * item_rate_card%
	// ---------------------------------------
	// Tiers separated by -1: floors 1-33 / 34-66 / 67-100
	setarray .@raw,
		1002,1113,1031,1063,1007,1019,1023,1015,1076,1041,1036,1242, -1,
		1163,1132,1144,1117,1192,1109,1253,1193,1219,1302,1504,1198, -1,
		1370,1374,1295,1216,1219,1504,1302,1198,1163,1132,1253,1117;
	.@tier = 1;
	for (.@i = 0; .@i < getarraysize(.@raw); .@i++) {
		if (.@raw[.@i] == -1) { .@tier++; continue; }
		if (getmonsterinfo(.@raw[.@i], MOB_NAME) == "null") continue;
		setd ".pool"+.@tier+"["+.pn[.@tier]+"]", .@raw[.@i];
		.pn[.@tier]++;
	}
	// main MVP per 10th floor (floors 10..90)
	setarray .mvp, 1086,1115,1087,1150,1147,1038,1039,1157,1373;
	// floor 100 finale
	setarray .@fin, 1708,1389,1751;
	// bonus MVPs (floors ending in 5, and the finale extra)
	setarray .@bon, 1159,1059,1046,1112,1190,1251,1252,1272,1511,1583,1312,1492,1623,1685,1785,1832,1719,1779,1871,1874;
	// mini bosses (Gilded)
	setarray .@min, 1096,1120,1388,1582,1088,1090,1091,1093;
	for (.@i = 0; .@i < getarraysize(.@fin); .@i++) if (getmonsterinfo(.@fin[.@i], MOB_NAME) != "null") .final[getarraysize(.final)] = .@fin[.@i];
	for (.@i = 0; .@i < getarraysize(.@bon); .@i++) if (getmonsterinfo(.@bon[.@i], MOB_NAME) != "null") .bonus[getarraysize(.bonus)] = .@bon[.@i];
	for (.@i = 0; .@i < getarraysize(.@min); .@i++) if (getmonsterinfo(.@min[.@i], MOB_NAME) != "null") .mini[getarraysize(.mini)] = .@min[.@i];
	// elemental stones used by alchemist: Flame Heart, Mystic Frozen, Rough Wind, Great Nature
	setarray .elem, 994,995,996,997;
	$@abt_run = 0;
	end;

OnStart:
	$@abt_begin = gettimetick(2);
	$@abt_end = $@abt_begin + .limit;
	$@abt_floor = 0;
	$@abt_left = 0;
	.busy = 0;
	killmonsterall .map$;
	mapannounce .map$, "The Green Hour begins. Floor 1 in a few seconds...", bc_map, 0xADFF2F;
	playbgmall "BGM\\absinthe_t1.mp3", .map$;
	initnpctimer;
	donpcevent "AbsintheCtl#aft::OnNext";
	end;

OnNext:
	if (!$@abt_run) end;
	if (.busy) end;
	.busy = 1;
	if ($@abt_floor > 0) callsub L_Clear;
	if ($@abt_floor >= .maxfloor) {
		callsub L_Win;
		.busy = 0;
		end;
	}
	sleep 5000;
	if (!$@abt_run) { .busy = 0; end; }
	$@abt_floor++;
	callsub L_Spawn;
	.busy = 0;
	end;

OnMobKill:
	if (!$@abt_run) end;
	$@abt_left--;
	callsub L_Drops, killedrid;
	if ($@abt_left <= 0) donpcevent "AbsintheCtl#aft::OnNext";
	end;

OnTimer10000:
	if (!$@abt_run) end;
	if (gettimetick(2) > $@abt_begin + 30 && getmapusers(.map$) < 1) { donpcevent "AbsintheCtl#aft::OnEnd"; end; }
	if (gettimetick(2) >= $@abt_end) {
		mapannounce .map$, "Time is up. The fairy sweeps you out.", bc_map, 0xFF6347;
		donpcevent "AbsintheCtl#aft::OnEnd";
		end;
	}
	initnpctimer;
	end;

OnEnd:
	stopnpctimer;
	mapwarp .map$, "SavePoint", 0, 0;
	killmonsterall .map$;
	$@abt_run = 0;
	$@abt_floor = 0;
	$@abt_left = 0;
	.busy = 0;
	end;

OnPCLoadMapEvent:
	if (strcharinfo(3) != .map$) end;
	if ($@abt_floor <= 1) abt_rev = 2;
	end;

OnPCDieEvent:
	if (strcharinfo(3) != .map$) end;
	if (abt_rev <= 0) { dispbottom "The fairy cannot bring you back again."; end; }
	abt_rev--;
	dispbottom "The Absinthe Fairy whispers: 'Not yet, dear...' ("+abt_rev+" revives left)";
	sleep2 3000;
	if (Hp == 0 && strcharinfo(3) == .map$) {
		atcommand "@alive";
		percentheal 100, 100;
	}
	end;

// ---- floor spawn ------------------------------------------------
L_Spawn:
	.@f = $@abt_floor;
	.@tier = (.@f <= 33) ? 1 : ((.@f <= 66) ? 2 : 3);
	.@n = min(.max_wave, 6 + .@f / 4);
	if (.@f == 1 || .@f == 34 || .@f == 67) playbgmall "BGM\\absinthe_t"+.@tier+".mp3", .map$;
	mapannounce .map$, "Floor "+.@f+"/"+.maxfloor, bc_map, 0xADFF2F;
	if (.@f == .maxfloor) {
		playbgmall "BGM\\absinthe_boss.mp3", .map$;
		mapannounce .map$, "FINAL FLOOR: the Philosopher's Throne!", bc_map, 0xFFD700;
		for (.@i = 0; .@i < getarraysize(.final); .@i++) callsub L_Mob, .final[.@i], 1;
		if (getarraysize(.bonus)) callsub L_Mob, .bonus[rand(getarraysize(.bonus))], 1;
		.@n = 10;
	} else if (.@f % 10 == 0) {
		playbgmall "BGM\\absinthe_boss.mp3", .map$;
		.@id = .mvp[.@f / 10 - 1];
		if (getmonsterinfo(.@id, MOB_NAME) == "null" && getarraysize(.bonus)) .@id = .bonus[rand(getarraysize(.bonus))];
		callsub L_Mob, .@id, 1;
		.@n = .@n / 3;
	} else if (.@f % 10 == 5 && getarraysize(.bonus)) {
		callsub L_Mob, .bonus[rand(getarraysize(.bonus))], 1;
		.@n = .@n / 2;
	} else if (.@f % 10 == 1 && .@f > 10) {
		playbgmall "BGM\\absinthe_t"+.@tier+".mp3", .map$;
	}
	if (.@f % 3 == 0 && .@f % 10 != 0 && getarraysize(.mini)) {
		.@k = 1 + (.@f > 66);
		for (.@i = 0; .@i < .@k; .@i++) callsub L_Mob, .mini[rand(getarraysize(.mini))], 2;
	}
	for (.@i = 0; .@i < .@n; .@i++)
		if (.pn[.@tier]) callsub L_Mob, getd(".pool"+.@tier+"["+rand(.pn[.@tier])+"]"), 0;
	return;

// getarg(0)=mob id, getarg(1)=0 normal / 1 mvp / 2 gilded mini
L_Mob:
	.@nm$ = "--en--";
	if (getarg(1) == 0) .@nm$ = "Transmuted "+getmonsterinfo(getarg(0), MOB_NAME);
	if (getarg(1) == 2) .@nm$ = "Gilded "+getmonsterinfo(getarg(0), MOB_NAME);
	.@gid = monster(.map$, 0, 0, .@nm$, getarg(0), 1, "AbsintheCtl#aft::OnMobKill");
	$@abt_left++;
	if (.@gid) {
		.@hp = getunitdata(.@gid, UMOB_MAXHP) * $@abt_hp / 100;
		if (getarg(1) == 2) .@hp = .@hp * 3;
		setunitdata .@gid, UMOB_MAXHP, .@hp;
		setunitdata .@gid, UMOB_HP, .@hp;
		setunitdata .@gid, UMOB_ATKMIN, getunitdata(.@gid, UMOB_ATKMIN) * $@abt_atk / 100;
		setunitdata .@gid, UMOB_ATKMAX, getunitdata(.@gid, UMOB_ATKMAX) * $@abt_atk / 100;
	}
	return;

// ---- floor clear rewards ----------------------------------------
L_Clear:
	.@f = $@abt_floor;
	.@rw = $@abt_rw;
	.@tok = 1;
	if (.@f % 10 == 0) .@tok = 10 + .@f / 10;
	.@tok = max(1, .@tok * .@rw / 100);
	if (.@f > $abt_rec) { $abt_rec = .@f; $abt_rec_name$ = $@abt_leader$; }
	mapannounce .map$, "Floor "+.@f+" cleared!", bc_map, 0x7CFC00;
	.@c = getunits(BL_PC, .@aid, false, .map$);
	for (.@i = 0; .@i < .@c; .@i++) {
		if (!attachrid(.@aid[.@i])) continue;
		getitem 32100, .@tok;
		getexp .@f * 6000 * .@rw / 100, .@f * 4500 * .@rw / 100;
		if (.@f > abt_best) abt_best = .@f;
		if (.@f % 10 == 0) callsub L_Milestone, .@f, .@rw;
		detachrid;
	}
	return;

L_Milestone:
	.@f = getarg(0);
	.@m = max(1, getarg(1) / 100);
	getitem 603, .@m;                                  // Old Blue Box
	if (.@f % 30 == 0) getitem 617, .@m;               // Old Purple Box
	if (.@f % 20 == 0) { getitem 7619, 2 * .@m; getitem 7620, 2 * .@m; }
	if (.@f == 50) getitem 32102, .@m;                 // Absinthe Fairy Box
	if (.@f >= 70) getitem 12103, .@m;                 // Bloody Branch
	if (.@f % 10 == 0) getitem 32105, 2 * .@m;         // Philosopher's Stone Fragments
	if (.@f == 100) {
		getitem 616, .@m; getitem 32103, .@m; getitem 32104, .@m;
		getitem 32106, .@m; getitem 32102, 3 * .@m; getitem 32100, 50 * .@m;
	}
	return;

L_Win:
	$abt_clears++;
	announce "[Absinthe Tower] "+$@abt_leader$+"'s party conquered the Philosopher's Throne!", bc_all, 0xFFD700;
	mapannounce .map$, "You are free to go. The tower closes in 30 seconds.", bc_map, 0xFFD700;
	playbgmall "BGM\\absinthe_victory.mp3", .map$;
	sleep 30000;
	donpcevent "AbsintheCtl#aft::OnEnd";
	return;

// ---- per-kill alchemical drops (in addition to the mob's normal drops) ---
// getarg(0) = killed mob id; killer is attached
L_Drops:
	if (!playerattached()) return;
	.@m = getarg(0);
	.@f = max(1, $@abt_floor);
	.@mvp = (getmonsterinfo(.@m, MOB_MVPEXP) > 0);
	.@mini = (inarray(.mini, .@m) >= 0);
	.@cr = getbattleflag("item_rate_card");
	if (.@cr < 100) .@cr = 100;
	.@k = (100 + .@f * 2) * $@abt_rw / 100 * .@cr / 100;   // percent multiplier
	// every monster
	callsub L_Roll, 32105, 300, 1, .@k;     // Philosopher's Stone Fragment 3%
	callsub L_Roll, .elem[rand(4)], 100, 1, .@k;   // an elemental stone 1%
	callsub L_Roll, 7619, 120, 1, .@k;      // Enriched Elunium
	callsub L_Roll, 7620, 120, 1, .@k;      // Enriched Oridecon
	callsub L_Roll, 616,   15, 1, .@k;      // Old Card Album 0.15%
	callsub L_Roll, 32103, 20, 1, .@k;      // Mini Boss Card Album 0.20%
	callsub L_Roll, 32104,  8, 1, .@k;      // MVP Card Album 0.08%
	callsub L_Roll, 617,   30, 1, .@k;      // Old Purple Box
	callsub L_Roll, 603,   60, 1, .@k;      // Old Blue Box
	callsub L_Roll, 32102, 50, 1, .@k;      // Absinthe Fairy Box
	if (.@mini) {
		callsub L_Roll, 32103, 3500, 1, .@k;
		callsub L_Roll, 32105, 6000, 3, .@k;
		callsub L_Roll, 616, 600, 1, .@k;
		callsub L_Roll, 617, 800, 1, .@k;
		callsub L_Roll, 32102, 1500, 1, .@k;
	}
	if (.@mvp) {
		callsub L_Roll, 32104, 2500, 1, .@k;
		callsub L_Roll, 32105, 10000, 5, .@k;
		callsub L_Roll, 616, 2500, 1, .@k;
		callsub L_Roll, 617, 4000, 1, .@k;
		callsub L_Roll, 32102, 10000, 1, .@k;
		callsub L_Roll, 32106, 400, 1, .@k;   // Philosopher's Stone 4%
		callsub L_Roll, .elem[rand(4)], 5000, 2, .@k;
	}
	return;

// getarg: 0 item, 1 base chance /10000, 2 amount, 3 multiplier %
L_Roll:
	.@c = min(10000, getarg(1) * getarg(3) / 100);
	if (rand(10000) >= .@c) return;
	if (getitemname(getarg(0)) == "null") return;
	if (checkweight(getarg(0), getarg(2)) == 0) return;
	getitem getarg(0), getarg(2);
	dispbottom "[Alchemy] The monster dropped "+getitemname(getarg(0))+" x"+getarg(2)+"!";
	return;
}

function<TAB>script<TAB>ABT_RollCard<TAB>{
	.@mode = getarg(0);
	if (.@mode == 1) {
		.@n = getarraysize(getvariableofnpc(.mini, "AbsintheCtl#aft"));
		copyarray .@pool[0], getvariableofnpc(.mini[0], "AbsintheCtl#aft"), .@n;
	} else {
		// MVP pool: the main + bonus + finale MVPs
		.@n = getarraysize(getvariableofnpc(.mvp, "AbsintheCtl#aft"));
		copyarray .@pool[0], getvariableofnpc(.mvp[0], "AbsintheCtl#aft"), .@n;
		.@b = getarraysize(getvariableofnpc(.bonus, "AbsintheCtl#aft"));
		copyarray .@pool[.@n], getvariableofnpc(.bonus[0], "AbsintheCtl#aft"), .@b;
		.@n += .@b;
		.@f = getarraysize(getvariableofnpc(.final, "AbsintheCtl#aft"));
		copyarray .@pool[.@n], getvariableofnpc(.final[0], "AbsintheCtl#aft"), .@f;
		.@n += .@f;
	}
	for (.@try = 0; .@try < 60 && .@n > 0; .@try++) {
		.@mob = .@pool[rand(.@n)];
		.@dc = getmobdrops(.@mob);
		for (.@j = 0; .@j < .@dc; .@j++) {
			.@it = $@MobDrop_item[.@j];
			if (getiteminfo(.@it, ITEMINFO_TYPE) == IT_CARD) {
				getitem .@it, 1;
				dispbottom "The album shimmers... "+getmonsterinfo(.@mob, MOB_NAME)+" Card!";
				return;
			}
		}
	}
	getitem 616, 1;   // fallback: Old Card Album
	return;
}
